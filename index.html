const express = require('express'); const axios = require('axios'); const app = express(); const PORT = 3000; // ڕێگەپێدانی CORS بۆ هەموو داواکارییەکان app.use((req, res, next) => { res.header('Access-Control-Allow-Origin', '*'); res.header('Access-Control-Allow-Headers', 'Origin, X-Requested-With, Content-Type, Accept'); next(); }); app.get('/api/getcode', async (req, res) => { try { const targetUrl = `http://124.221.92.205:56133/api/free/v2/getcode?id=319199&_t=${Date.now()}`; // ناردنی داواکاری ڕاستەوخۆ لە سێرڤەرەوە const response = await axios.get(targetUrl, { timeout: 5000 }); // ناردنەوەی وەڵامەکە بۆ Frontend res.json(response.data); } catch (error) { res.status(500).json({ error: 'نەتوانرا زانیاری لە سێرڤەرەوە وەربگیرێت', details: error.message }); } }); app.listen(PORT, () => { console.log(`سێرڤەرەکە کاردەکات لەسەر پۆڕتی http://localhost:${PORT}`); });
